-- Efeito de carregamento
local function showLoading()
    local loadingGui = Instance.new("ScreenGui")
    loadingGui.Name = "LoadingGui"
    loadingGui.ResetOnSpawn = false
    loadingGui.IgnoreGuiInset = true
    loadingGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    loadingGui.Parent = game:GetService("CoreGui")

    local bg = Instance.new("ImageLabel")
    bg.Size = UDim2.new(1, 0, 1, 0)
    bg.Position = UDim2.new(0, 0, 0, 0)
    bg.Image = "rbxassetid://14384894792" -- espada roxa
    bg.ImageTransparency = 0.2
    bg.BackgroundTransparency = 1
    bg.Parent = loadingGui

    local text = Instance.new("TextLabel")
    text.Size = UDim2.new(1, 0, 0, 100)
    text.Position = UDim2.new(0, 0, 0.5, -50)
    text.BackgroundTransparency = 1
    text.Text = "🔥 Carregando WL AFK Hub 🔥"
    text.TextColor3 = Color3.fromRGB(180, 0, 255)
    text.TextStrokeTransparency = 0
    text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    text.TextScaled = true
    text.Font = Enum.Font.FredokaOne
    text.Parent = loadingGui

    for i = 1, 10 do
        text.TextTransparency = 1 - (i * 0.1)
        task.wait(0.05)
    end

    wait(2.5)
    loadingGui:Destroy()
end
showLoading()

-- Funções de configuração
local function restoreDefaultSettings()
    OrionLib.Flags["fpsCap"].Value = 60
    OrionLib.Flags["antiLag"].Value = true
    OrionLib.Flags["saveConfig"].Value = true
    OrionLib.Flags["configFolder"].Value = "WLAfkHub"
end

local function applyGraphicsSettings()
    if OrionLib.Flags["antiLag"].Value then
        -- Desativa efeitos gráficos pesados
        game.Lighting.Bloom.Enabled = false
        game.Lighting.DepthOfField.Enabled = false
        game.Lighting.SunRays.Enabled = false
        game.Lighting.GlobalShadows.Enabled = false
    else
        -- Restaura efeitos gráficos
        game.Lighting.Bloom.Enabled = true
        game.Lighting.DepthOfField.Enabled = true
        game.Lighting.SunRays.Enabled = true
        game.Lighting.GlobalShadows.Enabled = true
    end
end
